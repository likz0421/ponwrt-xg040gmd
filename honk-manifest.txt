# honk 进固件的锁定参数（wrapper 包方案）
#
# 为什么不做源码构建：honk 的 eBPF 对象要 nightly Rust + 预编译 bpf-linker +
# BoringSSL(Go)，且 daeuniverse/openwrt-packages 里没有那套 CI 定义（只有
# build-vmlinux-btf.yml）；他们的 .apk 又是 ADBd 单文件格式，tar/gzip 解不开，
# 用构建里的 host apk 解又被 package/utils/lua[host] 编译失败挡住。
#
# 所以：做一个真正的 in-tree 包 package/custom/honk，
#   - 二进制取 honk 官方 release 资产（musl 静态、内嵌 doona），
#     由 buildroot 自己下载并按 PKG_HASH 校验；
#   - init / UCI / config.dae 从 daeuniverse 的 feed 源码里按 SHA 取；
#   - luci-app-honk / luci-lib-daeuniverse 也从同一 feed 按 SHA 取源码，
#     作为普通 LuCI 包编进来。
# 结果：honk 是固件里的正常包，进 apk 数据库、可 apk upgrade、依赖齐全。

HONK_VERSION=0.1.0-beta.4
HONK_ASSET=honk-core-v0.1.0-beta.4-aarch64-unknown-linux-musl-full.tar.gz
HONK_URL=https://github.com/daeuniverse/honk/releases/download/v0.1.0-beta.4/honk-core-v0.1.0-beta.4-aarch64-unknown-linux-musl-full.tar.gz
# 与 release 附带的 SHA256SUMS 逐字核对过（不是只信我自己下载后算的那次）
HONK_SHA256=121c94f3a30091c5550ae503ddf7bce118def08ddc301a7f8ffa740f956476d0
HONK_FILE=honk-core            # 资产内的文件名（静态链接 ELF，30.7MB）

# 三个包的来源：daeuniverse/openwrt-packages（feed 形态，源码），锁 SHA
DAEUNIV_URL=https://github.com/daeuniverse/openwrt-packages
